$c03PhaseClock=[Diagnostics.Stopwatch]::StartNew()
[Console]::Error.WriteLine('C03PHASE|script_started|' + $c03PhaseClock.Elapsed.TotalMilliseconds.ToString([Globalization.CultureInfo]::InvariantCulture))
$ErrorActionPreference='Stop'
$utf8=[System.Text.UTF8Encoding]::new($false)
[Console]::OutputEncoding=$utf8; $OutputEncoding=$utf8
$apps=@(Get-StartApps)
[Console]::Error.WriteLine('C03PHASE|start_apps_complete|' + $c03PhaseClock.Elapsed.TotalMilliseconds.ToString([Globalization.CultureInfo]::InvariantCulture))
$duplicateIds=@{}
$apps | Group-Object Name | Where-Object Count -GT 1 | ForEach-Object {
    $_.Group | ForEach-Object { $duplicateIds[$_.AppID]=$true }
}
$targets=@{}
[Console]::Error.WriteLine('C03PHASE|duplicates_grouped|' + $c03PhaseClock.Elapsed.TotalMilliseconds.ToString([Globalization.CultureInfo]::InvariantCulture))
if ($duplicateIds.Count -gt 0) {
    $shell=$null; $folder=$null
    try {
        $shell=New-Object -ComObject Shell.Application
        $folder=$shell.NameSpace('shell:AppsFolder')
        foreach ($item in $folder.Items()) {
            if ($duplicateIds.ContainsKey($item.Path)) {
                $targets[$item.Path]=$item.ExtendedProperty('System.Link.TargetParsingPath')
            }
        }
    } catch {
        # Missing Shell metadata preserves the original ambiguity.
        $targets=@{}
    } finally {
        if ($null -ne $folder) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($folder) }
        if ($null -ne $shell) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($shell) }
    }
}
[Console]::Error.WriteLine('C03PHASE|shell_metadata_complete|' + $c03PhaseClock.Elapsed.TotalMilliseconds.ToString([Globalization.CultureInfo]::InvariantCulture))
@($apps | ForEach-Object {
    [pscustomobject]@{ Name=$_.Name; AppID=$_.AppID; TargetPath=$targets[$_.AppID] }
}) | ConvertTo-Json -Compress
[Console]::Error.WriteLine('C03PHASE|serialization_complete|' + $c03PhaseClock.Elapsed.TotalMilliseconds.ToString([Globalization.CultureInfo]::InvariantCulture))
